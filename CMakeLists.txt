cmake_minimum_required(VERSION 3.20)
project(auto_swingshot LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
include(CTest)
if(BUILD_TESTING)
  add_executable(test_detection tests/test_detection.cpp)
  target_include_directories(test_detection PRIVATE src)
  add_test(NAME detection COMMAND test_detection)
endif()
option(BUILD_OPENRAC_MOD "Build native OpenRAC mod" ON)
if(BUILD_OPENRAC_MOD)
  if(NOT OpenRACMod_DIR)
    if(DEFINED OPENRAC_SDK)
      set(OpenRACMod_DIR "${OPENRAC_SDK}/cmake")
    else()
      set(OpenRACMod_DIR "${CMAKE_CURRENT_SOURCE_DIR}/.sdk/openrac-mod-sdk-1.14.5/cmake")
    endif()
  endif()
  find_package(OpenRACMod 1.0 REQUIRED)
  openrac_add_mod(auto_swingshot MANIFEST openrac-mod.json SOURCES src/mod.cpp)
  target_include_directories(auto_swingshot PRIVATE src)
endif()
